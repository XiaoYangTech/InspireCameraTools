using System.Windows;
using System.Windows.Data;
using Cvte.Windows.Localization;

namespace EasiCamera.UI;

public class ImageEnhanceDisplayToolBarItem : DisplayToolBarItem
{
	public ImageEnhanceDisplayToolBarItem()
	{
		base.SortHint = 101.0;
		base.FunctionModule = FunctionModule.MainWindow;
		Binding binding = new Binding
		{
			Path = new PropertyPath("Camera", null),
			Source = EC.CameraApi.DeviceViewModel.MainPreViewElement,
			Converter = new SeewoCameraButtonVisibilityConverter(),
			ConverterParameter = (SeewoCameraType.SC03 | SeewoCameraType.SC05 | SeewoCameraType.SC07 | SeewoCameraType.SC03C | SeewoCameraType.SC06 | SeewoCameraType.SC07E | SeewoCameraType.SC07F)
		};
		BindingOperations.SetBinding(this, DisplayToolBarItem.VisibilityProperty, binding);
		Binding binding2 = new Binding
		{
			Path = new PropertyPath("CurrentSlide", null),
			Source = EC.BoardApi.Board,
			Converter = new SlideToButtonDisabledConverter()
		};
		BindingOperations.SetBinding(this, UiItem.IsCanClickDisabledProperty, binding2);
	}

	protected override void OnNormalClick(object sender, RoutedEventArgs e)
	{
		base.OnNormalClick(sender, e);
		EC.EventReporter.ReportEvent("ie_enter");
		EnhanceModesDialog enhanceModesDialog = new EnhanceModesDialog();
		enhanceModesDialog.Owner = Application.Current.MainWindow;
		enhanceModesDialog.ShowDialog();
	}

	protected override void OnDisabledClick(object sender, RoutedEventArgs e)
	{
		base.OnDisabledClick(sender, e);
		EC.BoardApi.RaiseTryToSetSelectedItemVisible();
		EC.Notification.ShowInfoSmartly(Lang.Get("Lang.Info.EnhanceImageReturnToCamera"));
	}
}
